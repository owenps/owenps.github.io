{{- $classes := split (default "" .Attributes.class) " " -}} {{- $reveal := and .PlainText (in $classes "reveal") -}} {{- $wrapLeft := in $classes "wrap-left" -}} {{- $wrapRight := in $classes "wrap-right" -}} {{- if and $wrapLeft $wrapRight -}} {{- errorf "An image cannot use both wrap-left and wrap-right at %s" .Position -}} {{- end -}} {{- if and .IsBlock (or $reveal $wrapLeft $wrapRight) -}} {{- $figureClasses := slice -}} {{- if $reveal }}{{ $figureClasses = $figureClasses | append "image-reveal" }}{{ end -}} {{- if $wrapLeft }}{{ $figureClasses = $figureClasses | append "wrap-left" }}{{ end -}} {{- if $wrapRight }}{{ $figureClasses = $figureClasses | append "wrap-right" }}{{ end -}} {{- else -}} {{- end -}}
